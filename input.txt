# ---------------------------------------------------------------------
# Residue contacts (cavity_precompute.py)
# ---------------------------------------------------------------------
# A residue is in contact with CO2 if any atom of target_selection belonging
# to it lies within contact_cutoff_nm of any atom of ligand_selection.

# ---------------------------------------------------------------------
# First exit of CO2 from the protein (cavity_precompute.py)
# ---------------------------------------------------------------------
# First exit = first frame in which the minimum distance between ligand_selection
# and exit_selection exceeds exit_threshold_nm. The exit site is the set of
# residues of exit_selection within exit_contact_cutoff_nm of CO2 in the
# exit_window_ps preceding the exit.

# ---------------------------------------------------------------------

min_residues_cavity     = 3       # visits with fewer consensus residues are discarded
